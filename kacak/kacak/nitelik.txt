SELECT A.*,B.BAGIMSIZ_NO, B.BINA_NO FROM EML_BEYAN A, NUM_BAGIMSIZ_BOLUM_W B, NUM_ISITMA_TURU C, NUM_BINA_DETAY D
WHERE B.BINA_NO=D.BINA_NO
AND B.BAGIMSIZ_NO=D.BAGIMSIZ_NO
AND A.MAHALLE_KODU=B.MAHALLE_KODU
AND A.CADDE_SOKAK_KODU=B.CADDE_SOKAK_KODU
AND NVL(A.KAPI_NO,0)=NVL(B.KAPI_NO,0)
AND NVL(A.ALT_KAPI_NO,' ')=NVL(B.ALT_KAPI_NO,' ')
AND NVL(A.DAIRE_NO,0)=NVL(B.DAIRE_NO,0)
--AND NVL(A.ALT_DAIRE_NO,' ')=NVL(B.ALT_DAIRE_NO,' ')
AND A.BEYAN_TURU=1 
AND A.DONEM_KODU=2018 
AND A.SATIS_TARIHI IS NULL 
AND A.BAGIMSIZ_NO IS NULL
AND D.ISITMA_TURU=C.ISITMA_TURU
AND NVL(C.KALORIFER_EH,'H')='E'
AND NVL(A.BINA_KALORIFER,'H')='H'
