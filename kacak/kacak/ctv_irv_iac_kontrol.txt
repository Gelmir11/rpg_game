SELECT SUM(CTV_SAYISI) CTV_SAYISI,SUM(IRV_SAYISI) IRV_SAYISI,SUM(IAC_SAYISI) IAC_SAYISI FROM
(
 SELECT COUNT(*) CTV_SAYISI, 0 IRV_SAYISI, 0 IAC_SAYISI FROM NUM_CTV_IRV_IAC_KACAK WHERE MAHALLE_KODU IN (5,6,7) AND CTV_KACAK_EH = 'E'
 UNION ALL
 SELECT  0 CTV_SAYISI, COUNT(*) IRV_SAYISI, 0 IAC_SAYISI FROM NUM_CTV_IRV_IAC_KACAK WHERE MAHALLE_KODU IN (5,6,7) AND IRV_KACAK_EH = 'E'
 UNION ALL
 SELECT 0 CTV_SAYISI, 0 IRV_SAYISI, COUNT(*) IAC_SAYISI FROM NUM_CTV_IRV_IAC_KACAK WHERE MAHALLE_KODU IN (5,6,7) AND IAC_KACAK_EH = 'E')
