CREATE TABLE GOLCUK_KACAK_RAKAMLARI
( TURU                             VARCHAR2(2),
  BAGIMSIZ_NO                NUMBER(10),
  PARSEL_KODU                NUMBER(10),
  SICIL_KODU	         NUMBER(15),		
  BEYAN_TURU                  NUMBER(1),
  BEYAN_SIRASI                 NUMBER(5),
  TAHAKKUK_TOPLAMI      NUMBER(18,2),
  CEZA_TOPLAMI              NUMBER(18,2));
CREATE INDEX GOLCUK_KAC_BAG_IND ON GOLCUK_KACAK_RAKAMLARI (BAGIMSIZ_NO,SICIL_KODU,BEYAN_TURU,BEYAN_SIRASI);
CREATE INDEX GOLCUK_KAC_PRS_IND ON GOLCUK_KACAK_RAKAMLARI (PARSEL_KODU,SICIL_KODU,BEYAN_TURU,BEYAN_SIRASI);
CREATE INDEX GOLCUK_KAC_BEY_IND ON GOLCUK_KACAK_RAKAMLARI (SICIL_KODU,BEYAN_TURU,BEYAN_SIRASI);



SELECT TURU,SUM(TAHAKKUK_TOPLAMI),SUM(CEZA_TOPLAMI) 
  FROM GOLCUK_KACAK_RAKAMLARI
GROUP BY TURU
ORDER BY TURU
